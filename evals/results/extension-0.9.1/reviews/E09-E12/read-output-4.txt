18: def normalized(value):
19:     if isinstance(value, dict):
20:         return {key: normalized(item) for key, item in value.items()}
21:     if isinstance(value, list):
22:         return [normalized(item) for item in value]
23:     if isinstance(value, str):
24:         return '\n'.join(line.rstrip() for line in value.replace('\r\n', '\n').strip().splitlines())
25:     if type(value) is float and value.is_integer():
26:         return int(value)
27:     return value

30: def digest(value):
31:     raw = json.dumps(normalized(value), sort_keys=True, ensure_ascii=False,
32:                      separators=(',', ':'), allow_nan=False).encode('utf-8')
33:     return hashlib.sha256(raw).hexdigest()

36: def compile_handoff(plan):
37:     facts = {key: copy.deepcopy(plan[key]) for key in GLOBAL_KEYS if key in plan}
38:     header_facts = copy.deepcopy(facts)
39:     header_facts['ending_state'] = copy.deepcopy(plan['beats'][-1]['after'])
40:     scopes = [{'scope': 'headers', 'facts': header_facts, 'expression': copy.deepcopy(plan['headers'])}]
41:     beats = {b['id']: b for b in plan['beats']}
42:     preceding = []
43:     for section in plan['sections']:
44:         selected = [copy.deepcopy(beats[bid]) for bid in section['beat_ids']]

48:         preceding.extend({key: copy.deepcopy(beat[key]) for key in
